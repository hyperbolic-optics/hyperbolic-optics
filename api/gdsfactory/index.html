    gdsfactory Integration API¶

    gdsfactory ¶

    gdsfactory bridge, imported as hog: PDK stacks, periodic layouts and metasurfaces.

    Needs the gdsfactory extra and an active PDK; see the gdsfactory user guide.

    Modules:

    • geometry –

      Line gratings: a grid-snapped PCell, and extraction that refuses anything else.

    • lattice –

      2D-periodic layouts: grid-snapped PCells, exact extraction, PDK stacks and rcwa2d layers.

    • materials –

      Materials for gdsfactory stacks: specifications, crystal orientation, PDK names.

    • metasurface –

      Metasurfaces whose cells differ site by site: layout, read-back and SAX models.

    • simulate –

      Labelled results from a layout (simulate_*) or a hand-written stack (solve_*).

    • sparameters –

      Cached four-port S-parameters for PDK stacks and metasurface cells.

    • stack –

      Vertical stacks, written by hand or read from a PDK LayerStack.

    def half_space_permittivity(self, which: str, wavelength_um: Any) -> jax.Array:
    """Real scalar ε; invalid half-spaces raise eagerly and return NaN under JIT."""
    spec = self.incident if which == "incident" else self.exit
    tensor = material_tensor(spec, jnp.asarray(wavelength_um, dtype=float))
    eps = tensor[..., 0, 0]
    if isinstance(tensor, jax.core.Tracer):
        return jnp.where(_real_scalar(tensor, jnp), eps.real, jnp.nan)
    if not np.all(_real_scalar(tensor, np)):
        raise ValueError(
            f"The {which} half-space ({spec!r}) must be isotropic and lossless; the solvers "
            "need real, positive ε there. Put an absorbing or anisotropic material in a "
            "finite layer instead."
        )
    return eps.real

    def exit_medium(self, wavelength_um: Any) -> jax.Array:
    """The exit half-space for 2D-periodic solves (:mod:`~hyperbolic_optics.rcwa2d`).

    A real scalar ε where it is isotropic and lossless; otherwise its 3×3
    tensor -- a crystal, hyperbolic or absorbing substrate, whose two
    transmitted modes then replace p and s. Decided at ``wavelength_um``, or
    under ``jit`` at a representative wavelength of the material.
    """
    wavelength = jnp.asarray(wavelength_um, dtype=float)
    traced = isinstance(wavelength, jax.core.Tracer)
    spec = self.exit
    while isinstance(spec, Oriented):
        spec = spec.material
    if not traced:
        probe = wavelength
    elif isinstance(spec, str):
        probe = 2e4 / sum(dispersion.frequency_range(spec))  # mid-window, cm⁻¹ → µm
    else:
        probe = 1.55
    with jax.ensure_compile_time_eval():  # a static decision, even under jit
        tensor = np.asarray(material_tensor(self.exit, probe))
    if np.all(_real_scalar(tensor, np)):
        return (
            self.half_space_permittivity("exit", wavelength)
            if traced
            else tensor[..., 0, 0].real
        )
    return material_tensor(self.exit, wavelength)

    def extract_line_grating(
    component: gf.Component,
    layer: gf.typings.LayerSpec,
    tolerance_um: float | None = None,
) -> LineGrating:
    """Read a line grating back from the polygons of ``component`` on ``layer``.

    Every polygon must be an axis-aligned rectangle; all must span the same y
    range, have the same width, and start at equally spaced x. Anything else
    raises ``ValueError`` naming what failed, because a solver fed a guessed
    geometry returns a confident, wrong answer.

    Args:
        component: A gdsfactory component, e.g. from ``gf.import_gds``.
        layer: The layer the ridges are drawn on.
        tolerance_um: Allowed deviation; defaults to one database unit.

    Returns:
        The :class:`LineGrating`, with the period from a least-squares fit to the
        ridge positions.
    """
    tolerance = component.kcl.dbu * 1.01 if tolerance_um is None else tolerance_um
    layer_index = gf.get_layer(layer)
    polygons = component.get_polygons_points(layers=[layer])
    points = polygons.get(layer_index, [])
    if len(points) < 2:
        raise ValueError(f"need at least two ridges on layer {layer}, found {len(points)}")

    rectangles = []
    for polygon in points:
        polygon = np.asarray(polygon, dtype=float)
        xs, ys = np.unique(np.round(polygon[:, 0], 9)), np.unique(np.round(polygon[:, 1], 9))
        if len(polygon) != 4 or len(xs) != 2 or len(ys) != 2:
            raise ValueError(
                f"polygon on layer {layer} is not an axis-aligned rectangle: {polygon}"
            )
        rectangles.append((xs[0], xs[1], ys[0], ys[1]))
    rectangles.sort()
    x0, x1, y0, y1 = (np.array(column) for column in zip(*rectangles, strict=True))

    if np.ptp(y0) > tolerance or np.ptp(y1) > tolerance:
        raise ValueError("ridges do not all span the same y range; not a line grating")
    widths = x1 - x0
    if np.ptp(widths) > tolerance:
        raise ValueError(f"ridge widths differ by {np.ptp(widths):.4g} µm; not a line grating")
    index = np.arange(len(x0))
    period, start = np.polyfit(index, x0, 1)
    if np.max(np.abs(x0 - (start + period * index))) > tolerance:
        raise ValueError("ridges are not equally spaced; not a periodic line grating")
    if np.mean(widths) >= period - tolerance:  # the fitted period carries rounding
        raise ValueError("ridges touch or overlap; no gap to form a grating")
    return LineGrating(
        period_um=float(period),
        fill=float(np.mean(widths) / period),
        offset=float((start / period) % 1.0),
        n_periods=len(x0),
        length_um=float(np.mean(y1 - y0)),
        origin_um=(float(x0[0]), float(np.mean(y0))),
    )

    def extract_unit_cell(
    component: gf.Component,
    layer: gf.typings.LayerSpec,
    period_um: tuple[float, float] | None = None,
    lattice_um: tuple[tuple[float, float], tuple[float, float]] | None = None,
) -> UnitCell:
    """Read a 2D-periodic pattern back from the polygons of ``component`` on ``layer``.

    The polygons are grouped into translated copies of one another. Each group
    must fill the same complete ``n₁ × n₂`` lattice, so every cell holds the
    same motif. Shapes must not overlap, even across cell edges, and must not
    have holes. Anything else raises ``ValueError``, because a solver fed a
    guessed geometry returns a confident, wrong answer.

    Args:
        component: A gdsfactory component, e.g. from ``gf.import_gds``.
        layer: The layer the pattern is drawn on.
        period_um: A rectangular lattice's period.
        lattice_um: Any lattice's primitive vectors ``((a1x, a1y), (a2x, a2y))``.
            Without either, the lattice is inferred -- rectangular, hexagonal or
            oblique, motifs that repeat a shape (a dimer) included: the
            shortest basis under which the drawing is one motif on a complete
            parallelogram of cells. Pass it to impose a lattice instead.
    """
    dbu = component.kcl.dbu
    polygons = _hulls(kf.kdb.Region(component.begin_shapes_rec(gf.get_layer(layer))))
    if len(polygons) < 2:
        raise ValueError(
            f"need a repeated pattern on layer {layer}, found {len(polygons)} polygons"
        )

    groups: dict[tuple, list[np.ndarray]] = {}
    for points in polygons:
        groups.setdefault(_shape_key(points), []).append(points)
    corners = {key: np.array([p.min(axis=0) for p in members]) for key, members in groups.items()}

    if lattice_um is None and period_um is not None:
        lattice_um = ((period_um[0], 0.0), (0.0, period_um[1]))
    if lattice_um is not None:
        bases = [
            tuple(np.round(np.asarray(v, dtype=float) / dbu).astype(np.int64) for v in lattice_um)
        ]
    else:
        reference = max(corners.values(), key=len)
        bases = _candidate_bases(reference)
        if not bases:
            raise ValueError(
                "cannot infer the lattice: the pattern does not repeat in two directions. "
                "Pass period_um or lattice_um, or draw at least two cells each way."
            )

    found, overlapping = None, None
    for a1, a2 in bases:
        split = _complete_lattices(groups, corners, a1, a2)
        if split is None:
            continue
        counts, motif = split
        origin = np.min([first for _, first in motif], axis=0)
        drawn = {(_shape_key(points), tuple(points.min(axis=0))) for points in polygons}
        rebuilt = {
            (_shape_key(points), tuple(first + i * a1 + j * a2))
            for points, first in motif
            for i in range(counts[0])
            for j in range(counts[1])
        }
        if drawn != rebuilt or len(polygons) != len(motif) * counts[0] * counts[1]:
            continue
        if _overlaps(tuple(points - origin for points, _ in motif), a1, a2):
            overlapping = overlapping or (a1, a2)
            continue
        found = (a1, a2, counts, motif, origin)
        break
    if found is None and overlapping is not None:
        a1, a2 = overlapping
        raise ValueError(
            f"shapes on layer {layer} overlap (within a cell or across its edges) on the lattice "
            f"{tuple(a1 * dbu)}, {tuple(a2 * dbu)} µm; the solver needs non-overlapping shapes"
        )
    if found is None:
        described = "the given lattice" if lattice_um is not None else "any lattice inferred"
        raise ValueError(
            f"the pattern on layer {layer} does not fill a complete lattice at {described}: it "
            "is not one motif repeated on a periodic array. If a cell repeats a shape, pass "
            "period_um or lattice_um"
        )
    a1, a2, counts, motif, origin = found
    cell_polygons = tuple((points - origin) for points, _ in motif)

    rectangular = a1[1] == 0 and a2[0] == 0
    return UnitCell(
        period_um=(a1[0] * dbu, a2[1] * dbu) if rectangular else None,
        polygons=tuple(p.astype(float) * dbu for p in cell_polygons),
        n_cells=counts,
        origin_um=(float(origin[0] * dbu), float(origin[1] * dbu)),
        lattice_um=None if rectangular else (tuple(a1 * dbu), tuple(a2 * dbu)),
    )

    Each z interval uses its own mask polygons, including derived layers.
    Lower mesh_order wins; WAFER covers the footprint. Uncovered regions are
    ambient. The lowest interval must be uniform and becomes the substrate.
    A StackLayer can represent two materials per interval; more are refused.
    Crystal cuts come from ``materials``, as in :func:`vertical_stack`.
    ``sidewall_slices=N`` approximates each nonvertical level by N midpoint
    slices, respecting ``width_to_z`` and scalar ``bias`` (µm per edge).

    walls. Generic ``z_to_bias`` and directional biases are refused when slicing.

        raise ValueError("sidewall_slices must be a nonnegative integer.")
    a1, a2 = np.asarray(lattice_um, float)
    # one rounded corner plus whole lattice steps, so the footprint tiles the plane exactly
    step1, step2 = _steps(a1, a2, dbu)

    footprint = kf.kdb.Region(
        kf.kdb.Polygon(
            [kf.kdb.Point(*map(int, start + s)) for s in (0 * step1, step1, step1 + step2, step2)]
        )
    )
    if footprint.area() == 0:
        raise ValueError("The lattice footprint must have nonzero area on the layout grid.")
    present = []
    for name, level in layer_stack.layers.items():
        if not level.thickness:
            continue

            level.layer.layer
        ) == gf.get_layer("WAFER")
        # shapes crossing the footprint's edge wrap around, as RCWA periodises them
        periodic = None if wafer else _tiled(level.layer.get_shapes(component), step1, step2)
        region = footprint if wafer else periodic & footprint
        if not region.is_empty():

            if wafer:
                present.append((low, high, name, level, region))
                continue
            if not sidewall_slices:
                _warn_process_geometry(name, level)
                present.append((low, high, name, level, region))

                raise ValueError(

                raise ValueError(f"Invalid sidewall angle for level {name!r}.")

                np.linspace(low, high, sidewall_slices + 1) if level.sidewall_angle else [low, high]
            )
            reference_z = level.zmin + level.width_to_z * level.thickness
            altitude = _altitude(a1, a2)
            for lo, hi in zip(edges[:-1], edges[1:], strict=True):
                offset = (level.bias or 0.0) + (reference_z - (lo + hi) / 2) * np.tan(
                    np.radians(level.sidewall_angle)
                )
                if abs(offset) >= altitude / 2:
                    raise ValueError(f"PDK level {name!r}: offset exceeds half the cell altitude.")
                sliced = periodic.sized(round(offset / dbu)) & footprint
                # Keep even an empty slice: its z interval still contains background.
                present.append((float(lo), float(hi), name, level, sliced))
    if not present:
        raise ValueError("No PDK material is present in the cell footprint.")
    edges = sorted({z for low, high, *_ in present for z in (low, high)})

    def spec(name):
        return ambient if name is None else resolve_material(name, materials)

    layers, substrate = [], None
    for low, high in zip(edges[:-1], edges[1:], strict=True):
        active = sorted(
            (p for p in present if p[0] <= low and p[1] >= high), key=lambda p: _mesh_order(p[3])
        )
        occupied, regions, ties = kf.kdb.Region(), {}, {}
        for _, _, name, level, region in active:
            priority = _mesh_order(level)
            for other_material, other in ties.get(priority, []):
                if other_material != level.material and not (region & other).is_empty():
                    raise ValueError(f"PDK materials tie on mesh_order near level {name!r}.")
            ties.setdefault(priority, []).append((level.material, region))
            visible = region - occupied
            if not visible.is_empty():
                regions[level.material] = regions.get(level.material, kf.kdb.Region()) | visible
            occupied = occupied | region
        uncovered = footprint - occupied
        if not uncovered.is_empty():
            regions[None] = uncovered
        if substrate is None:
            if len(regions) != 1:
                raise ValueError("The lowest PDK interval must be a uniform substrate.")
            substrate = spec(next(iter(regions)))
            continue
        if len(regions) > 2:
            raise ValueError(f"More than two materials in the cell at z={low:g}–{high:g} µm.")
        label = f"z={low:g}–{high:g}"
        if len(regions) == 1:
            layers.append(StackLayer(high - low, spec(next(iter(regions))), name=label))
            continue
        background = None if None in regions else max(regions, key=lambda k: regions[k].area())
        material = next(k for k in regions if k != background)
        region = regions[material].merged()
        # Trapezoids represent holes without discarding their interiors. All
        # pieces have the same material, so their common edges cancel in RCWA.
        if any(p.holes() for p in region.each()):
            region = region.decompose_trapezoids_to_region()
        polygons = tuple(_vertices_um(region, dbu))
        layers.append(
            StackLayer(
                high - low,
                spec(material),
                background=spec(background),
                polygons=polygons,
                name=label,
            )
        )
    return VerticalStack(tuple(reversed(layers)), incident=ambient, exit=substrate)

    def model_from_npz(filepath: Path | str | Mapping[str, np.ndarray]):
    """A SAX model from saved S-parameters, interpolated over wavelength, modes kept.

    Returns ``model(wl) → SDict`` keyed ``("port@mode", "port@mode")``. Real and
    imaginary parts are interpolated linearly (``jnp.interp``), as
    ``gplugins.sax.read.model_from_npz`` does, which instead drops the mode.
    """
    data = _load(Path(filepath)) if not isinstance(filepath, Mapping) else dict(filepath)
    x = jnp.asarray(data["wavelengths"])
    order = jnp.argsort(x)
    x = x[order]
    entries = {
        tuple(key.split(",")): jnp.asarray(value)[order]
        for key, value in data.items()
        if key != "wavelengths"
    }

    def model(wl=1.55):
        wl = jnp.asarray(wl, dtype=float)
        return {
            ports: jnp.interp(wl, x, jnp.real(v)) + 1j * jnp.interp(wl, x, jnp.imag(v))
            for ports, v in entries.items()
        }

    return model

    def grating_stack(
    component: gf.Component,
    grating_layer: gf.typings.LayerSpec,
    layer_stack: LayerStack,
    materials: Mapping[str, MaterialSpec] | None = None,
    ambient: MaterialSpec = 1.0,
    samples: int = 3,
) -> tuple[LineGrating, VerticalStack]:
    """A line grating and the patterned stack its etch makes in the PDK's LayerStack.

    The stack is sampled at ``samples`` points across a ridge and across a gap,
    in two periods; every ridge sample must agree, and every gap sample, or the
    layout is not a single line grating in this stack and a ``ValueError`` says
    so. Where the ridge and gap columns differ, the layer is patterned.
    """
    grating = extract_line_grating(component, grating_layer)
    period, width = grating.period_um, grating.fill * grating.period_um
    x0, y0 = grating.origin_um
    y = y0 + grating.length_um / 2
    periods = sorted({0, grating.n_periods // 2})

    def column(x_in_period, span):
        profiles = [
            tuple(vertical_profile(component, layer_stack, (x0 + k * period + x_in_period(i), y)))
            for k in periods
            for i in range(samples)
        ]
        if len(set(profiles)) != 1:
            raise ValueError(
                f"The stack varies within the {span} of the grating on {grating_layer}: "
                "another layer is patterned differently there, so this is not a single "
                "line grating in this layer stack."
            )
        return list(profiles[0])

    ridge = column(lambda i: width * (i + 0.5) / samples, "ridges")
    gap = column(lambda i: width + (period - width) * (i + 0.5) / samples, "gaps")

    return grating, patterned_stack(ridge, gap, materials, ambient)
